# ទំនួលខុសត្រូវពីគ្រោះថ្នាក់ការជែងនៅតៃវ៉ាន់ វិនិច្ឆ័យយ៉ាងណា?

![រូបភាពពន្យល់ការវិនិច្ឆ័យទំនួលខុសត្រូវពីគ្រោះថ្នាក់ការជែងនៅតៃវ៉ាន់ និងលំដាប់ជែងដែលមានសុវត្ថិភាព](../images/012-taiwan-overtaking-accident-liability/featured-01.jpg)

ពេលយានខាងមុខរត់យឺត ការជែង (超車) អាចមើលទៅដូចជាជម្រើសធម្មតា ប៉ុន្ដែពិតជាអំពើបើកបរដែលមានហានិភ័យមិនតិច។ មុនចាប់ផ្ដើមជែង ត្រូវពិនិត្យជាមួយគ្នាអំពីផ្នែកផ្លូវ ស្ថានភាពចរាចរណ៍ទាំងពីរទិស ចលនារបស់យានខាងមុខ និងថាមានទីកន្លែងគ្រប់ដើម្បីជែងឆ្លងដោយសុវត្ថិភាព រួចត្រឡប់ចូលផ្លូវដើម (車道) ឬទេ។

## លក្ខខណ្ឌនៃការជែងតាមមាត្រា 101 នៃបទប្បញ្ញត្ដិសុវត្ថិភាពចរាចរណ៍ផ្លូវគោក

[មាត្រា 101 នៃបទប្បញ្ញត្ដិសុវត្ថិភាពចរាចរណ៍ផ្លូវគោកតៃវ៉ាន់ (道路交通安全規則)](https://law.moj.gov.tw/LawClass/LawSingle.aspx?pcode=K0040013&flno=101) កំណត់ទាំងលក្ខខណ្ឌដែលហាមជែង និងលំដាប់ដែលត្រូវអនុវត្ដពេលអនុញ្ញាតឱ្យជែងក្នុងផ្លូវតែមួយ។

មាត្រា 101 ហាមជែងនៅផ្នែកផ្លូវដែលមានសញ្ញាបង្ហាញផ្លូវកោង ជម្រាលចោត ស្ពានតូច ផ្លូវរូងក្រោមដី ឬផ្លូវប្រសព្វ ព្រមទាំងនៅផ្លូវរថភ្លើងកាត់ (鐵路平交道) និងតំបន់សំណង់ផ្លូវ។

លើសពីនេះ នៅទីកន្លែង ឬផ្នែកផ្លូវដែលមានសញ្ញាសាលា ឬមន្ទីរពេទ្យ នៅកន្លែងមានសញ្ញា ឬសម្គាល់ហាមជែងផ្សេង នៅពេលមានយានមកពីទិសផ្ទុយ ឬនៅពេលយានខាងមុខរត់តគ្នាពីរ ឬច្រើនគ្រឿង ក៏មិនអាចជែងបានដែរ។

អ្នកបើកបរយានខាងក្រោយដែលចង់ជែងយានខាងមុខក្នុងផ្លូវតែមួយ ត្រូវបន្លឺស្នែងខ្លីពីរដង ឬភ្លឹបភ្លែតពន្លឺមុខមួយដងជាមុន។ មិនត្រូវប្រើស្នែង ឬពន្លឺមុខម្ដងហើយម្ដងទៀត ដើម្បីបង្ខំឱ្យយានខាងមុខផ្ដល់ផ្លូវឡើយ។

យានខាងក្រោយអាចជែងបាន លុះត្រាតែយានខាងមុខបានបន្ថយល្បឿន និងចៀសទៅចំហៀង ឬបានបង្ហាញចេតនាផ្ដល់ផ្លូវដោយសញ្ញាដៃ ឬដោយបើកភ្លើងបត់ស្ដាំ (方向燈)។

បន្ទាប់មក យានដែលជែងត្រូវបើកភ្លើងបត់ឆ្វេង ហើយឆ្លងទៅខាងឆ្វេង ដោយរក្សាចន្លោះពីយានខាងមុខមិនតិចជាង 0.5 ម៉ែត្រ។ ពេលបានចន្លោះសុវត្ថិភាពហើយ ទើបបើកភ្លើងបត់ស្ដាំ និងត្រឡប់ចូលផ្លូវដើមដោយសុវត្ថិភាព។

លំដាប់សញ្ញានិងការផ្ដល់ផ្លូវក្នុងផ្លូវតែមួយនេះ មិនមានន័យថាអនុញ្ញាតឱ្យជែងនៅទីកន្លែង ឬក្នុងលក្ខខណ្ឌដែលហាមឡើយ។ មុនចាប់ផ្ដើមជែង នៅតែត្រូវវិនិច្ឆ័យលក្ខខណ្ឌទាំងអស់តាមមាត្រា 101 ជាមួយគ្នា។

## ឧទាហរណ៍គ្រោះថ្នាក់អនាមិកដែលការិយាល័យបានទទួលធ្វើ

ក្នុងសំណុំរឿងអនាមិកដែលការិយាល័យនេះបានទទួលធ្វើ អ្នកបើកបរម៉ូតូ (機車) A មានអ្នកដំណើរ B អង្គុយជាមួយ កំពុងរត់លើផ្លូវភ្នំ។ ខាងមុខមានរថយន្ដគ្រួសារ (小客車) ពីរគ្រឿង។ យានទី 1 ដែលនៅមុខបំផុតរត់យឺត ធ្វើឱ្យយានទី 2 និងម៉ូតូរត់យឺតតាម។

![ផែនទីគ្រោះថ្នាក់បង្ហាញផ្លូវជែងរបស់ម៉ូតូ និងរថយន្ដពីរគ្រឿងខាងមុខ លើផ្លូវភ្នំ](../images/012-taiwan-overtaking-accident-liability/img-01.jpg)

A ព្យាយាមជែងយានទាំងពីរក្នុងពេលតែមួយ ដូច្នេះចូលផ្លូវទិសផ្ទុយ (對向車道) និងបង្កើនល្បឿន។ ក្នុងពេលជាមួយគ្នា យានទី 2 ក៏ត្រៀមជែងយានទី 1 ដែរ ដោយបើកភ្លើងបត់រួចចូលផ្លូវទិសផ្ទុយក្នុងរយៈពេលមិនដល់ 1 វិនាទី។ ម៉ូតូខ្វះពេលនិងចន្លោះគ្រប់ដើម្បីហ្វ្រាំង ដូច្នេះបុកនឹងយានទី 2។

B រងរបួសក្បាលធ្ងន់ និងស្លាប់នៅកន្លែងកើតហេតុ។ A បាត់ស្មារតី និងត្រូវបាននាំទៅមន្ទីរពេទ្យ។

ដំបូង គ្រួសារ A និង B យល់ថាការប្ដូរផ្លូវភ្លាមរបស់យានទី 2 ជាមូលហេតុសំខាន់នៃការបុក។ រឿងនេះបានចូលដល់នីតិវិធីតុលាការ ហើយក្នុងដំណើរនោះមានការវាយតម្លៃជំនាញអំពីគ្រោះថ្នាក់ (鑑定) ច្រើនដង។

តាមលទ្ធផលវាយតម្លៃ A ត្រូវបានវិនិច្ឆ័យថាទទួលខុសត្រូវសំខាន់ក្នុងការបុកនេះ។ សេចក្ដីសន្និដ្ឋាននោះកំណត់ត្រឹមអង្គហេតុនៃសំណុំរឿងនេះប៉ុណ្ណោះ។ ការវាយតម្លៃបានពិចារណាជាមួយគ្នាទាំងការដែល A ព្យាយាមជែងយានខាងមុខពីរគ្រឿងដែលរត់តគ្នា ការចូលផ្លូវទិសផ្ទុយ ការប្រើល្បឿនដែលធ្វើឱ្យខ្វះពេលនិងចន្លោះហ្វ្រាំង ការមិនបានផ្ដល់សញ្ញាស្នែង ឬភ្លឹបភ្លែតពន្លឺមុខតាមដែលកំណត់ ចលនាប្ដូរផ្លូវរបស់យានទី 2 រចនាសម្ព័ន្ធផ្លូវនិងផ្លូវរត់ និងភស្ដុតាងផ្សេងដែលបានរក្សា។

ការមានលទ្ធផលវាយតម្លៃតាមសំណុំរឿងបែបនេះ មិនមានន័យថាការខកខានសញ្ញាតាមដែលកំណត់តែមួយដង នឹងកំណត់ទំនួលខុសត្រូវជានិច្ចឡើយ។ កំហុសក្នុងគ្រោះថ្នាក់ការជែងនៅតែខុសគ្នាតាមទីកន្លែងកើតហេតុ រចនាសម្ព័ន្ធផ្លូវ ល្បឿន ចលនាយាន សញ្ញាតាមដែលកំណត់ ចន្លោះពេល ភាពមើលឃើញ និងភស្ដុតាងផ្សេង។

## ចំណុចដែលគួរពិនិត្យពេលវិនិច្ឆ័យទំនួលខុសត្រូវពីគ្រោះថ្នាក់

មុនជែង ត្រូវពិនិត្យជាមុនថាទីកន្លែង ឬស្ថានភាពចរាចរណ៍ចូលករណីហាមជែងឬទេ។ បើជាករណីដែលអនុវត្ដលំដាប់សម្រាប់ផ្លូវតែមួយ ត្រូវផ្ដល់សញ្ញាស្នែង ឬភ្លឹបភ្លែតពន្លឺមុខតាមដែលកំណត់ ដោយមិនបង្ខំឱ្យយានខាងមុខផ្ដល់ផ្លូវ រង់ចាំការបង្ហាញចេតនាផ្ដល់ផ្លូវឱ្យច្បាស់ បន្ទាប់មកទើបបញ្ចប់ការជែងដោយរក្សាចន្លោះគ្រប់ និងទុកចន្លោះសម្រាប់ត្រឡប់ចូលផ្លូវដើមដោយសុវត្ថិភាព។

ការគោរពមាត្រា 101 មានសារៈសំខាន់ ប៉ុន្ដែដោយខ្លួនវាមិនធានាថានឹងចៀសផុតគ្រោះថ្នាក់ ហើយមិនកំណត់លទ្ធផលនៃការវាយតម្លៃ ឬនៃសំណុំរឿងក្រោយជាមុនឡើយ។ អ្នកបើកបរនៅតែត្រូវទុកចន្លោះសុវត្ថិភាពបន្ថែម និងឆ្លើយតបតាមស្ថានភាពផ្លូវខាងមុខ។

ផ្នែកឯកសារបន្ថែម អាចមើល [រូបពន្យល់វិធាននិងលំដាប់ការជែង](https://gonews.com.tw/car/daily/21934/)។ ឯកសារនេះជាប្រភពទីពីរ ដូច្នេះពេលពិនិត្យលក្ខខណ្ឌផ្លូវច្បាប់ សូមមើលបទប្បញ្ញត្ដិផ្លូវការដែលអនុវត្ដបច្ចុប្បន្នផងដែរ។
