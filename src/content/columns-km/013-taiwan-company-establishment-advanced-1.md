# ការបង្កើតក្រុមហ៊ុននៅតៃវ៉ាន់ - ភាគជ្រៅ - 1

![រូបភាពសំខាន់](../images/013-taiwan-company-establishment-advanced-1/featured-01.jpg)

ក្នុងរយៈពេលថ្មីៗនេះ មានមនុស្សជាច្រើនកំពុងរៀបចំបង្កើតក្រុមហ៊ុននៅតៃវ៉ាន់។ វីសាធ្វើការ និងលិខិតស្នាក់នៅជនបរទេស (外僑居留證, ARC) ជានីតិវិធីដាច់ដោយឡែក មិនកើតដោយស្វ័យប្រវត្ដិពីការបង្កើតក្រុមហ៊ុនឡើយ។

នាងខ្ញុំសូមប្រមូលសំណួរដែលមនុស្សជាច្រើនចង់ដឹង ក្នុងដំណាក់កាលបង្កើតក្រុមហ៊ុន មករៀបចំទុកជាមួយគ្នា។

លោកអ្នកដែលបានអាន [**មូលដ្ឋាននៃការបង្កើតក្រុមហ៊ុន**](/km/columns/taiwan-company-establishment-basics) រួចហើយ សូមអានភាគជ្រៅដែលមានព័ត៌មានលម្អិតបន្ថែមនេះបន្ដ។

សំណួរនិងចម្លើយខាងក្រោមអាចជាគោលទូទៅសម្រាប់លោកអ្នកដែលកំពុងពិចារណាបង្កើតក្រុមហ៊ុននៅតៃវ៉ាន់។

![](../images/013-taiwan-company-establishment-advanced-1/img-01.jpg)

**1. ចង់បង្កើតក្រុមហ៊ុន ប៉ុន្ដែមិនទាន់រកឃើញ អាសយដ្ឋានចុះបញ្ជីក្រុមហ៊ុន តើបង្កើតបានដែរឬទេ**។

ពេលជនបរទេសនឹងបង្កើតក្រុមហ៊ុននៅតៃវ៉ាន់ ត្រូវដាក់លិខិតផែនការវិនិយោគ (投資計畫書) ទៅនាយកដ្ឋានពិនិត្យការវិនិយោគ ក្រសួងសេដ្ឋកិច្ច (經濟部投資審議司) ដើម្បីទទួលការពិនិត្យ។

នាយកដ្ឋានពិនិត្យការវិនិយោគត្រូវពិនិត្យថាដើមទុននោះត្រូវបានប្រើសម្រាប់គោលបំណងវិនិយោគពិតឬអត់។

ត្រូវពិនិត្យថាមិនមែនជាករណីភាគីពីបរទេសនាំទុនចូល ក្រោមលេសផ្សេង។

ពេលសរសេរលិខិតផែនការវិនិយោគ មិនចាំបាច់សរសេរអាសយដ្ឋានក្រុមហ៊ុន ឱ្យត្រឹមត្រូវជាក់លាក់ទេ គ្រាន់តែសរសេរតំបន់ក៏គ្រប់ហើយ (ឧទាហរណ៍ ក្រុងតៃប៉ិ (臺北市)) ​។

ទោះយ៉ាងណា ក្រោយមកពេលបើកគណនីការិយាល័យត្រៀមបង្កើតក្រុមហ៊ុន (公司籌備處帳戶) នៅធនាគារ ធនាគារជាច្រើននឹងពិនិត្យ កិច្ចសន្យាជួល (租賃契約) និងចេញទៅពិនិត្យទីតាំងក្រុមហ៊ុននៅនឹងកន្លែង។

បុគ្គលិកធនាគារតៃវ៉ាន់ម្នាក់ បាននិយាយដោយផ្ទាល់ទៅនាងខ្ញុំដូចតទៅ។

ក្នុងរយៈពេលថ្មីៗនេះ មានករណីជនបរទេសជាច្រើន បើកគណនីរួចរត់ចេញ ការបើកគណនីក៏ត្រូវបានពិនិត្យកាន់តែតឹងរ៉ឹង។

ក្នុងអត្ថបទមូលដ្ឋានមុននេះ អ្នកអានម្នាក់បាននិយាយថា ការបើកគណនីធនាគារជាដំណាក់កាលពិបាកបំផុត ហើយនាងខ្ញុំក៏យល់ស្របដែរ។

នៅតៃវ៉ាន់មានករណីលាងលុយ (洗錢) ច្រើន ដូច្នេះការបើកគណនីក៏ត្រូវបានពិនិត្យកាន់តែតឹងរ៉ឹង។

**ដំបូន្មន់:**

ការពិនិត្យផែនការវិនិយោគក៏ត្រូវការពេលដែរ។

គួរធ្វើការសរសេរលិខិតផែនការវិនិយោគ ទន្ទឹមនឹងការស្វែងរក អាសយដ្ឋានចុះបញ្ជីក្រុមហ៊ុន។

តាមមាត្រា 9 នៃច្បាប់ស្ដីពីការវិនិយោគរបស់ជនបរទេស (外國人投資條例) ទុនចូលរួមវិនិយោគដែលបានអនុម័ត ត្រូវមកដល់គ្រប់ចំនួន ក្នុងរយៈពេលដែលស្ថាប័នមានសមត្ថកិច្ចកំណត់ (核定期限) ហើយរយៈពេលពិតត្រូវពិនិត្យពី ឯកសារអនុម័តនីមួយៗ (ក្នុងការអនុវត្ដច្រើនតែកំណត់ប្រហែល 1 ឆ្នាំ)។

ដូច្នេះនៅមានពេលគ្រប់ដើម្បីរកទីតាំងក្រុមហ៊ុន ចុះកិច្ចសន្យាជួល។

បន្ទាប់មកបើកគណនីការិយាល័យត្រៀមបង្កើតក្រុមហ៊ុន (公司籌備處帳戶) នៅធនាគារ និងផ្ទេរដើមទុនចូលមក។

​

**2. ទោះគ្មានលិខិតស្នាក់នៅជនបរទេស (外僑居留證, ARC) នៅតៃវ៉ាន់ តើបើកគណនីក្រុមហ៊ុននៅធនាគារបានដែរឬទេ**។

អាចធ្វើបាន។

តាមធម្មតាធនាគារសុំឯកសារបញ្ជាក់អត្ដសញ្ញាណពីរប្រភេទ។

បើគ្មានលិខិតស្នាក់នៅជនបរទេស (外僑居留證, ARC) អាចដាក់ពាក្យសុំ **តារាងព័ត៌មានមូលដ្ឋាននៃលេខសម្គាល់បង្រួបបង្រួម** (統一證號基本資料表) នៅនាយកដ្ឋានអន្ដោប្រវេសន៍ ក្រសួងមហាផ្ទៃតៃវ៉ាន់ (內政部移民署)។

ឯកសារអាចចេញក្នុងថ្ងៃតែមួយបាន។

ប៉ុន្ដែ នាយកដ្ឋានអន្ដោប្រវេសន៍ ក្រសួងមហាផ្ទៃតៃវ៉ាន់ (內政部移民署) មានអ្នកមកទាក់ទងច្រើនណាស់ ដូច្នេះគួរទៅព្រឹកដើម្បីយកលេខជួរ។

​

**3. បានឮថាពេលពិនិត្យផែនការវិនិយោគ ត្រូវសរសេរកម្រិតសិក្សា និងប្រវត្ដិការងារ ប៉ុន្ដែបារម្ភថាកម្រិតសិក្សា និងប្រវត្ដិការងាររបស់ខ្លួន នឹងមិនត្រូវនឹងប្រភេទអាជីវកម្មនៃក្រុមហ៊ុនដែលនឹងបង្កើត**។

នាយកដ្ឋានពិនិត្យការវិនិយោគ ពិនិត្យប្រវត្ដិរបស់វិនិយោគិន។

ប៉ុន្ដែការពិនិត្យមិនតឹងរ៉ឹងពេកឡើយ។

អ្នកដាក់ពាក្យអាចលើកបទពិសោធន៍ការងារ រួមទាំងការងារក្រៅម៉ោង ច្រើនប្រភេទ។

អ្នកដាក់ពាក្យពន្យល់ឱ្យលម្អិតគ្រប់ ដើម្បីឱ្យអ្នកពិនិត្យយល់តាម។

ចំណុចនេះគួរពិភាក្សាជាមួយមេធាវីតៃវ៉ាន់ឱ្យគ្រប់។

​

**4. ពេលជួលទីកន្លែងសម្រាប់ចុះបញ្ជីក្រុមហ៊ុន (ឧទាហរណ៍ ហាងភោជនីយដ្ឋាន) ត្រូវប្រុងប្រយ័ត្នអ្វីខ្លះ**។

ដោយការបង្កើតក្រុមហ៊ុនចំណាយពេលប្រហែល **3 ខែ** ហើយការទទួល លិខិតអនុញ្ញាតការងារ (工作許可) និង លិខិតស្នាក់នៅជនបរទេស (外僑居留證, ARC) ក៏ចំណាយពេលប្រហែល **1 ខែ**។

ដូច្នេះគួរកំណត់ថ្ងៃចាប់ផ្ដើមកិច្ចសន្យា ឱ្យយឺតតាមដែលអាចធ្វើបាន។

លើសពីនេះ ក្នុងកិច្ចសន្យាជួលហាងនៅតៃវ៉ាន់ អ្នកឱ្យជួលច្រើនតែផ្ដល់ «**រយៈពេលតុបតែង**» (裝潢期間) ដល់អ្នកជួល។

រយៈពេលនេះជារយៈពេលលើកលែងថ្លៃជួល ដូច្នេះគួរចរចាអំពីចំណុចនេះ។

ការដែលជនបរទេសជួលហាងមិនងាយទេ ហើយប្រាក់ធានាជួលនៅតៃវ៉ាន់ស្ថិតក្នុងកម្រិតទាប (តាមធម្មតាស្មើថ្លៃជួល 2 ខែ) អ្នកឱ្យជួលទទួលការការពារតិច។

អ្នកឱ្យជួលច្រើននាក់ដូច្នេះស្ទាក់ស្ទើរ មិនចង់ឱ្យជនបរទេសជួល។

ដូច្នេះសូមណែនាំឱ្យចាប់ផ្ដើមស្វែងរកទីតាំងអាជីវកម្មតាំងពីដើម។

បើការចរចាជាមួយអ្នកឱ្យជួលមិនរលូន អាចស្នើវិធានបន្ថែម។

បើចាំបាច់ អាចស្នើឱ្យមាន ការផ្ទៀងផ្ទាត់ដោយសារការី (公證) លើកិច្ចសន្យាជួល ឬស្នើដាក់ប្រាក់ធានាបន្ថែម។

​

**5. ពេលបង្កើតក្រុមហ៊ុន តើអាចជួលទីកន្លែងការិយាល័យពាណិជ្ជបានដែរឬទេ**។

ខុសគ្នាតាម ប្រភេទអាជីវកម្ម (營業項目) ដែលចុះបញ្ជីរបស់ក្រុមហ៊ុន។

ឧទាហរណ៍ បើប្រភេទអាជីវកម្មដែលចុះបញ្ជីជាអាជីវកម្មភោជនីយដ្ឋាន ប៉ុន្ដែចង់ជួលទីកន្លែងការិយាល័យពាណិជ្ជ ការបើកគណនីធនាគារនឹងពិបាកណាស់។

ដូចដែលបាននិយាយខាងលើ នៅធនាគារ ធនាគារជាច្រើននឹងពិនិត្យ កិច្ចសន្យាជួល (租賃契約) និងចេញទៅពិនិត្យទីតាំងក្រុមហ៊ុននៅនឹងកន្លែង។

លើសពីនេះ នៅដំណាក់កាលចុះបញ្ជីក្រុមហ៊ុនចុងក្រោយ រដ្ឋាភិបាលក្រុង (市政府) ក៏កំណត់ឱ្យអាសយដ្ឋានចុះបញ្ជី ស្ថិតក្នុងតំបន់ដែលអាចធ្វើ អាជីវកម្មភោជនីយដ្ឋាន។

បើជាទីកន្លែងការិយាល័យពាណិជ្ជ មិនអាចចុះបញ្ជីក្រុមហ៊ុនបានឡើយ។

ឧទាហរណ៍ បើប្រភេទអាជីវកម្មអាចធ្វើនៅទីកន្លែងការិយាល័យពាណិជ្ជបាន ដូចជាអាជីវកម្មផ្សាយពាណិជ្ជ ឬអាជីវកម្មលក់ដុំ ក៏អាចបង្កើតក្រុមហ៊ុននៅទីកន្លែងបែបនោះបាន។ គួរពិនិត្យឱ្យច្បាស់ មុនចុះហត្ថលេខាលើកិច្ចសន្យាជួល។

សូមធ្វើការពិនិត្យនោះមុនចុះហត្ថលេខា។

*បើលោកអ្នកត្រូវការជំនួយ ឬការពិគ្រោះយោបល់អំពីច្បាប់តៃវ៉ាន់ សូមប្រើទំព័រទាក់ទងការិយាល័យ។ ការិយាល័យមិនសន្យារយៈពេលឆ្លើយតបឡើយ។ ការពិគ្រោះយោបល់ជាមួយមេធាវីធ្វើតែជាភាសាអង់គ្លេស ភាសាចិន (中文) ភាសាជប៉ុន និងភាសាកូរ៉េ*
